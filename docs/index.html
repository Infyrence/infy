Quick Example

Build a chain in 3 lines. No config files. No boilerplate.

python
from infy import tool, create_agent
from infy.providers.openai import OpenAIChat

@tool
def search(query: str) -> str:
    """Search the web."""
    return "sunny today"

agent = create_agent(
    OpenAIChat("gpt-4o"),
    tools=[search],
    system_prompt="You are helpful.",
)

result = agent("What's the weather?")
print(result.response.text)
