Streaming

Stream responses token-by-token with chunk merging and async iteration.

AIMessageChunk Merge

AIMessageChunk supports additive merging via the + operator. Content concatenates, tool call chunks merge by index:

python
from infy.messages import AIMessageChunk, ToolCallChunk

# Text streaming
chunks = [
    AIMessageChunk(content="Hello"),
    AIMessageChunk(content=" world"),
    AIMessageChunk(content="!"),
]

full = chunks[0]
for chunk in chunks[1:]:
    full = full + chunk

print(full.text)  # "Hello world!"

# Tool call streaming
chunk1 = AIMessageChunk(
    tool_call_chunks=[ToolCallChunk(name="search", args='{"q', index=0)]
)
chunk2 = AIMessageChunk(
    tool_call_chunks=[ToolCallChunk(args='uery": "test"}', index=0)]
)

merged = chunk1 + chunk2
print(merged.tool_call_chunks[0].args)  # '{"query": "test"}'

# Convert to full AIMessage
msg = merged.materialize()
print(msg.tool_calls[0].name)  # "search"

Stream Utilities

stream_text

Consume a stream and return the full accumulated text:

python
from infy.streaming import stream_text

# Works with any stream of chunks
stream = model.stream([HumanMessage(content="Tell me a story")])
full_text = stream_text(stream)
print(full_text)

stream_to_list

Consume a stream into a list of chunks:

python
from infy.streaming import stream_to_list

chunks = stream_to_list(model.stream([HumanMessage(content="Hello")]))
print(f"Got {len(chunks)} chunks")

merge_chunks

Merge a list of AIMessageChunk objects into one:

python
from infy.streaming import merge_chunks

chunks = [
    AIMessageChunk(content="Hello"),
    AIMessageChunk(content=" world"),
]
full = merge_chunks(chunks)
print(full.text)  # "Hello world"

tee

Split a stream into multiple independent copies:

python
from infy.streaming import tee

stream = model.stream([HumanMessage(content="Hello")])
stream1, stream2 = tee(stream, 2)

# Each can be consumed independently
text1 = stream_text(stream1)
text2 = stream_text(stream2)
# Both produce the same result

Async Streaming

All providers support async streaming with astream:

python
import asyncio
from infy.providers.openai import OpenAIChat
from infy.messages import HumanMessage

model = OpenAIChat("gpt-4o")

async def stream_response():
    async for chunk in model.astream([HumanMessage(content="Tell me a joke")]):
        print(chunk.text, end="", flush=True)

asyncio.run(stream_response())

Streaming with Runnables

Runnables support stream and astream:

python
from infy import Lambda, Sequence
from infy.providers.openai import OpenAIChat

model = OpenAIChat("gpt-4o")

# Sync streaming
for chunk in model.stream([HumanMessage(content="Hello")]):
    print(chunk.text, end="")

# Async streaming
async for chunk in model.astream([HumanMessage(content="Hello")]):
    print(chunk.text, end="")

Streaming in Agents

Agents support streaming via the underlying model:

python
from infy.providers.openai import OpenAIChat
from infy.messages import HumanMessage

model = OpenAIChat("gpt-4o")

# Stream a single model call
for chunk in model.stream([
    HumanMessage(content="What is 2+2?"),
]):
    print(chunk.text, end="", flush=True)

Complete Example

python
import asyncio
from infy.providers.openai import OpenAIChat
from infy.messages import HumanMessage, AIMessageChunk
from infy.streaming import stream_text, stream_to_list, merge_chunks

model = OpenAIChat("gpt-4o")

# 1. Sync streaming
print("=== Sync Stream ===")
for chunk in model.stream([HumanMessage(content="Tell me a short joke")]):
    print(chunk.text, end="", flush=True)
print()

# 2. Collect all chunks
print("=== Collect Chunks ===")
chunks = stream_to_list(model.stream([HumanMessage(content="One word answer: Python")]))
print(f"Got {len(chunks)} chunks")

# 3. Accumulate text
print("=== Accumulate ===")
full_text = stream_text(model.stream([HumanMessage(content="Say hello")]))
print(f"Full text: {full_text}")

# 4. Merge chunks into AIMessage
print("=== Merge ===")
full_msg = merge_chunks(chunks)
print(f"Final message: {full_msg.text}")

# 5. Async streaming
print("=== Async ===")
async def main():
    async for chunk in model.astream([HumanMessage(content="Hi")]):
        print(chunk.text, end="", flush=True)
    print()

asyncio.run(main())
