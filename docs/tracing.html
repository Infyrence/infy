Tracing

Pluggable observability with zero lock-in. Implement the Tracer protocol for any backend.

Tracer Protocol

Implement three methods to plug in any observability backend:

MethodSignatureDescription
on_start(name, input, metadata?) → strCalled when a traced function starts. Returns a run ID.
on_end(run_id, output, duration_ms) → NoneCalled on successful completion.
on_error(run_id, error, duration_ms) → NoneCalled when an exception is raised.

NoOpTracer

The default tracer does nothing. Zero overhead when tracing is not configured:

python
from infy.tracing import NoOpTracer, get_tracer

# Default tracer is NoOpTracer
tracer = get_tracer()
print(type(tracer))  # <class 'infy.tracing.NoOpTracer'>

@traced Decorator

Trace any function (sync or async) with the @traced decorator:

python
from infy.tracing import traced

@traced
def my_function(x: int) -> int:
    return x * 2

@traced(name="custom_name")
def another(x: int) -> int:
    return x + 1

# Works with async functions too
@traced
async def async_func(x: int) -> int:
    await asyncio.sleep(0.1)
    return x * 3

set_tracer / get_tracer

Set a global tracer that applies to all @traced functions:

python
from infy.tracing import set_tracer, get_tracer

# Set a custom tracer
set_tracer(my_custom_tracer)

# Later, get the current tracer
tracer = get_tracer()

Custom Tracer Example

Implement a tracer that logs to stdout:

python
import time
import uuid
from infy.tracing import set_tracer

class LoggingTracer:
    """Simple tracer that logs to stdout."""

    def on_start(self, name: str, input: Any, metadata: dict | None = None) -> str:
        run_id = str(uuid.uuid4())[:8]
        print(f"[{run_id}] START {name}")
        return run_id

    def on_end(self, run_id: str, output: Any, duration_ms: float) -> None:
        print(f"[{run_id}] END ({duration_ms:.1f}ms)")

    def on_error(self, run_id: str, error: Exception, duration_ms: float) -> None:
        print(f"[{run_id}] ERROR ({duration_ms:.1f}ms): {error}")

# Set it globally
set_tracer(LoggingTracer())

# Now all @traced functions will log
@traced
def process_data(data: str) -> str:
    return data.upper()

result = process_data("hello")
# Output:
# [a1b2c3d4] START process_data
# [a1b2c3d4] END (0.1ms)

OpenTelemetry Example

python
from opentelemetry import trace
from infy.tracing import set_tracer

tracer = trace.get_tracer("infy-app")

class OTelTracer:
    """OpenTelemetry integration."""

    def on_start(self, name, input, metadata=None):
        span = tracer.start_span(name)
        return str(id(span))

    def on_end(self, run_id, output, duration_ms):
        # End the span
        pass

    def on_error(self, run_id, error, duration_ms):
        # Record exception on span
        pass

set_tracer(OTelTracer())

Complete Example

python
import time
import uuid
from infy.tracing import traced, set_tracer, get_tracer

class MetricsTracer:
    """Collect timing metrics."""

    def __init__(self):
        self.calls = []

    def on_start(self, name, input, metadata=None):
        run_id = str(uuid.uuid4())[:8]
        self.calls.append({"name": name, "run_id": run_id, "start": time.time()})
        return run_id

    def on_end(self, run_id, output, duration_ms):
        for call in self.calls:
            if call["run_id"] == run_id:
                call["duration_ms"] = duration_ms
                call["status"] = "ok"
                break

    def on_error(self, run_id, error, duration_ms):
        for call in self.calls:
            if call["run_id"] == run_id:
                call["duration_ms"] = duration_ms
                call["status"] = "error"
                call["error"] = str(error)
                break

# Setup
metrics = MetricsTracer()
set_tracer(metrics)

@traced
def process(x: int) -> int:
    time.sleep(0.01)
    return x * 2

@traced(name="batch_process")
def batch(items: list) -> list:
    return [process(item) for item in items]

# Run
result = batch([1, 2, 3, 4, 5])
print(result)  # [2, 4, 6, 8, 10]

# Inspect metrics
for call in metrics.calls:
    print(f"{call['name']}: {call.get('duration_ms', 0):.1f}ms ({call.get('status', 'ok')})")
