RAG

Retrieval-Augmented Generation: retrieve relevant documents, augment the prompt, and generate answers.

Documents

Document is the unit of retrieval and indexing:

python
from infy.documents import Document, DocumentChunk

doc = Document(
    page_content="infy is a fast LLM framework with 5,103 lines of code.",
    metadata={"source": "docs/intro.md", "page": 1},
)

print(doc.id)       # auto-generated UUID
print(doc.text)     # page_content (alias)
print(doc.metadata) # {'source': 'docs/intro.md', 'page': 1}

# DocumentChunk preserves position info for reconstruction
chunk = DocumentChunk(
    document=doc,
    chunk_index=0,
    total_chunks=3,
    start_char=0,
    end_char=100,
)
print(chunk.page_content)  # from parent document
print(chunk.metadata)      # merged with chunk position info

Embeddings

Convert text to vectors for similarity search:

python
from infy.embeddings import FakeEmbeddings, OpenAIEmbeddings, OllamaEmbeddings

# For testing (deterministic, no API calls)
embeddings = FakeEmbeddings(dimension=128)
vectors = embeddings.embed_documents(["hello", "world"])
query_vec = embeddings.embed_query("hello")

# OpenAI (requires openai package)
embeddings = OpenAIEmbeddings(model="text-embedding-3-small")

# Ollama (local, requires ollama package)
embeddings = OllamaEmbeddings(model="nomic-embed-text")

VectorStore

Store and search document embeddings. InMemoryVectorStore uses Rust SIMD cosine similarity:

python
from infy.documents import Document
from infy.embeddings import FakeEmbeddings
from infy.vectorstores import InMemoryVectorStore

embeddings = FakeEmbeddings(dimension=128)
store = InMemoryVectorStore()

# Add documents with pre-computed embeddings
docs = [
    Document(page_content="infy is fast", metadata={"source": "readme.md"}),
    Document(page_content="LangChain is popular", metadata={"source": "comparison.md"}),
    Document(page_content="Python is great for AI", metadata={"source": "blog.md"}),
]
vectors = embeddings.embed_documents([d.page_content for d in docs])
store.add_documents(docs, vectors)

# Similarity search
query_vec = embeddings.embed_query("fast framework")
results = store.similarity_search(query_vec, k=2)

for doc, score in results:
    print(f"Score: {score:.3f} | {doc.page_content}")

print(f"Total docs: {store.count()}")

Retriever

The retriever connects embeddings to the vector store:

python
from infy.retrievers import VectorStoreRetriever, MultiQueryRetriever
from infy.embeddings import FakeEmbeddings
from infy.vectorstores import InMemoryVectorStore
from infy.documents import Document

# Setup
embeddings = FakeEmbeddings()
store = InMemoryVectorStore()
docs = [
    Document(page_content="infy has 5,103 lines of code"),
    Document(page_content="LangChain has 564,418 lines"),
    Document(page_content="infy uses Rust SIMD acceleration"),
]
vectors = embeddings.embed_documents([d.page_content for d in docs])
store.add_documents(docs, vectors)

# Basic retriever
retriever = VectorStoreRetriever(store=store, embeddings=embeddings, k=2)
results = retriever.get_relevant_documents("How many lines of code?")
print(results)

# Multi-query retriever (uses LLM to expand queries)
multi_retriever = MultiQueryRetriever(
    retriever=retriever,
    model=None,  # or provide a ChatModel
    num_queries=3,
)

Text Splitter

Chunk documents for embedding. Three strategies:

python
from infy.text_splitter import (
    TextSplitter,
    RecursiveTextSplitter,
    MarkdownTextSplitter,
)
from infy.documents import Document

# Simple character-based splitting
splitter = TextSplitter(chunk_size=500, chunk_overlap=50)
chunks = splitter.split_text("Long document text here...")
doc_chunks = splitter.split_documents([Document(page_content="Long text...")])

# Recursive (preserves structure better)
splitter = RecursiveTextSplitter(chunk_size=1000, chunk_overlap=200)
chunks = splitter.split_text("Document with headings\n\nParagraph 1\n\nParagraph 2")

# Markdown-aware (respects headings)
splitter = MarkdownTextSplitter(chunk_size=1000, chunk_overlap=200)
chunks = splitter.split_text("# Title\n\nContent here\n\n## Section\n\nMore content")

RAGChain

The complete pipeline: retrieve → build context → generate:

python
from infy.rag import RAGChain
from infy.retrievers import VectorStoreRetriever
from infy.embeddings import FakeEmbeddings
from infy.vectorstores import InMemoryVectorStore
from infy.documents import Document
from infy.providers.openai import OpenAIChat

# 1. Build the knowledge base
docs = [
    Document(page_content="infy is a 5,103 LOC LLM framework with Rust SIMD acceleration."),
    Document(page_content="infy supports OpenAI, Anthropic, Ollama, and Gemini providers."),
    Document(page_content="infy has 207 tests that run in 0.63 seconds."),
    Document(page_content="infy uses the pipe operator for chain composition: prompt | model | parser."),
]

embeddings = FakeEmbeddings()
store = InMemoryVectorStore()
vectors = embeddings.embed_documents([d.page_content for d in docs])
store.add_documents(docs, vectors)

# 2. Create retriever
retriever = VectorStoreRetriever(store=store, embeddings=embeddings, k=2)

# 3. Create RAG chain
rag = RAGChain(
    retriever=retriever,
    model=OpenAIChat("gpt-4o"),
    k=2,
    system_prompt=(
        "Answer the question based on the provided context.\n"
        "If the context doesn't contain enough information, say so.\n\n"
        "Context:\n{context}"
    ),
)

# 4. Query
result = rag.invoke("How many lines of code does infy have?")
print(result.response)
print(f"Sources: {[d.metadata.get('source') for d in result.source_documents]}")
print(f"Usage: {result.usage}")

Complete Pipeline Example

python
from infy.documents import Document
from infy.embeddings import FakeEmbeddings
from infy.vectorstores import InMemoryVectorStore
from infy.retrievers import VectorStoreRetriever
from infy.text_splitter import RecursiveTextSplitter
from infy.rag import RAGChain
from infy.providers.openai import OpenAIChat

# 1. Load and split documents
raw_docs = [
    Document(page_content=open(f).read(), metadata={"source": f})
    for f in ["doc1.md", "doc2.md", "doc3.md"]
]

splitter = RecursiveTextSplitter(chunk_size=1000, chunk_overlap=200)
chunks = splitter.split_documents(raw_docs)

# 2. Embed and store
embeddings = FakeEmbeddings()
store = InMemoryVectorStore()
vectors = embeddings.embed_documents([c.page_content for c in chunks])
store.add_documents(chunks, vectors)

# 3. Create RAG chain
retriever = VectorStoreRetriever(store=store, embeddings=embeddings, k=4)
rag = RAGChain(
    retriever=retriever,
    model=OpenAIChat("gpt-4o"),
    k=4,
)

# 4. Query
result = rag.invoke("What is the architecture of this project?")
print(result.response)
for doc in result.source_documents:
    print(f"  Source: {doc.metadata.get('source')}")
