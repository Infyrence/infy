Graphs

Build stateful, multi-step workflows with checkpointing, human-in-the-loop, and time-travel. ~1,000 lines vs LangGraph's 28,000+.

StateGraph Builder

Define your state schema, add nodes, connect edges, and compile:

python
from typing import TypedDict, Annotated
from infy.graph import StateGraph, START, END

def add_messages(left: list, right: list) -> list:
    return left + right

class State(TypedDict):
    messages: Annotated[list, add_messages]
    next_node: str

graph = StateGraph(State)
graph.add_node("agent", agent_fn)
graph.add_node("tools", tools_fn)
graph.add_edge(START, "agent")
graph.add_conditional_edges("agent", route_fn, {"tools": "tools", END: END})
graph.add_edge("tools", "agent")

app = graph.compile()

Builder API

MethodDescription
add_node(name, func)Add a processing node
add_edge(source, target)Unconditional edge
add_conditional_edges(source, condition, path_map)Conditional routing
compile(checkpointer, interrupt_before, interrupt_after)Build executable graph

CompiledGraph Execution

python
from infy.graph import StateGraph, START, END

# Basic execution
result = app.invoke({"messages": [HumanMessage(content="Hi")]})

# With config (for checkpointing)
result = app.invoke(
    {"messages": [HumanMessage(content="Hi")]},
    config={"configurable": {"thread_id": "thread-1"}},
)

Channels

Channels control how state values are updated between nodes:

LastValue

Stores the last value written. One write per step. Used for plain state fields:

python
class State(TypedDict):
    name: str      # LastValue(str) — one value
    count: int     # LastValue(int) — one value

BinOp

Reducer accumulation. Multiple writes are combined with a binary operator:

python
from typing import Annotated

class State(TypedDict):
    total: Annotated[int, operator.add]     # BinOp — sums values
    messages: Annotated[list, add_messages]  # BinOp — extends list

Topic

List accumulation. Per-step or across steps:

python
from infy.graph.channels import Topic

class State(TypedDict):
    events: Annotated[list, Topic(accumulate=True)]

Checkpointing

Save and restore graph state across invocations:

python
from infy.graph import InMemorySaver, StateGraph, START, END

# Create checkpointer
saver = InMemorySaver()

# Compile with checkpointer
app = graph.compile(checkpointer=saver)

# First invocation — creates checkpoint
result = app.invoke(
    {"messages": [HumanMessage(content="Hi")]},
    config={"configurable": {"thread_id": "thread-1"}},
)

# Resume — continues from last checkpoint
result2 = app.invoke(
    {"messages": [HumanMessage(content="What did I say?")]},
    config={"configurable": {"thread_id": "thread-1"}},
)

# Inspect state
state = app.get_state(config={"configurable": {"thread_id": "thread-1"}})
print(state.values)
print(state.next)  # nodes that would fire next

Conditional Routing

Route to different nodes based on state:

python
from infy.graph import StateGraph, START, END

def should_continue(state):
    """Route based on whether the last message has tool calls."""
    messages = state.get("messages", [])
    last = messages[-1]
    if hasattr(last, "tool_calls") and last.tool_calls:
        return "tools"
    return END

graph = StateGraph(State)
graph.add_node("agent", agent_fn)
graph.add_node("tools", tools_fn)
graph.add_edge(START, "agent")
graph.add_conditional_edges("agent", should_continue, {
    "tools": "tools",
    END: END,
})
graph.add_edge("tools", "agent")

Send for Fan-out

Dispatch dynamic tasks to named nodes:

python
from infy.graph.types import Send

def fan_out(state):
    """Create a Send for each item."""
    items = state.get("items", [])
    return [Send("process", {"item": item}) for item in items]

graph.add_conditional_edges("router", fan_out)

Command for Navigation

Return a Command from a node to jump or update state:

python
from infy.graph.types import Command

def my_node(state):
    # Jump directly to another node
    return Command(goto="special_node")

def update_state(state):
    # Update state values
    return Command(update={"count": state.get("count", 0) + 1})

Interrupt / Resume

Pause execution for human-in-the-loop:

python
from infy.graph import StateGraph, InMemorySaver

# Compile with interrupt
app = graph.compile(
    checkpointer=InMemorySaver(),
    interrupt_before=["tools"],  # Pause before tool execution
)

# First run — will interrupt at "tools" node
try:
    result = app.invoke(
        {"messages": [HumanMessage(content="Search for infy")]},
        config={"configurable": {"thread_id": "thread-1"}},
    )
except GraphInterrupt as e:
    print(f"Interrupted: {e.interrupts}")
    # Human reviews and provides value
    # Resume by invoking again with same thread_id

get_state / update_state

Inspect and modify state for time-travel debugging:

python
from infy.graph.types import StateSnapshot

# Get current state
state = app.get_state(config)
print(state.values)   # Current state dict
print(state.next)     # Nodes that would fire next
print(state.metadata) # {'step': 3}

# Update state manually (creates new checkpoint)
new_config = app.update_state(
    config,
    values={"messages": [HumanMessage(content="Override")]},
    as_node="agent",
)

# Now resume from the updated state
result = app.invoke({}, config=new_config)

create_graph_agent

Prebuilt agent with checkpointing and human-in-the-loop:

python
from infy.graph import create_graph_agent, InMemorySaver
from infy import tool
from infy.providers.openai import OpenAIChat

@tool
def search(query: str) -> str:
    """Search the web."""
    return f"Results for: {query}"

agent = create_graph_agent(
    OpenAIChat("gpt-4o"),
    tools=[search],
    system_prompt="You are helpful.",
    checkpointer=InMemorySaver(),
    interrupt_before=["tools"],
)

# Run with thread persistence
result = agent.invoke(
    {"messages": [{"role": "user", "content": "Search for infy"}]},
    config={"configurable": {"thread_id": "thread-1"}},
)

Complete Example

python
from typing import TypedDict, Annotated
from infy.graph import StateGraph, START, END, InMemorySaver
from infy import tool, create_graph_agent
from infy.providers.openai import OpenAIChat

@tool
def search(query: str) -> str:
    """Search the web."""
    return f"Results for: {query}"

@tool
def calculator(expression: str) -> str:
    """Evaluate a math expression."""
    return str(eval(expression))

# Create a full agent with checkpointing
agent = create_graph_agent(
    OpenAIChat("gpt-4o"),
    tools=[search, calculator],
    system_prompt="You are a helpful assistant.",
    checkpointer=InMemorySaver(),
    interrupt_before=["tools"],
)

# Run
config = {"configurable": {"thread_id": "thread-1"}}
result = agent.invoke(
    {"messages": [{"role": "user", "content": "What is 42 * 17?"}]},
    config=config,
)

# Inspect
state = agent.get_state(config)
print(f"State: {state.values}")
print(f"Next: {state.next}")

# Time-travel: update state and re-run
agent.update_state(config, {"messages": []})
result2 = agent.invoke({}, config=config)
